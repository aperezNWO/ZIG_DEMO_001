ZIG_VERSION="0.17.1"
curl -L "https://ziglang.org/download/${ZIG_VERSION}/zig-x86_64-linux-${ZIG_VERSION}.tar.xz" -o zig.tar.xz
tar -xf zig.tar.xz
sudo mv "zig-x86_64-linux-${ZIG_VERSION}" /usr/local/zig
sudo ln -sf /usr/local/zig/zig /usr/local/bin/zig
rm zig.tar.xz
zig version

.devcontainer/devcontainer.json

{
  "name": "Zig Backend",
  "image": "mcr.microsoft.com/devcontainers/base:ubuntu",
  "postCreateCommand": "ZIG_VERSION=0.17.1 && curl -fsSL https://ziglang.org/download/${ZIG_VERSION}/zig-x86_64-linux-${ZIG_VERSION}.tar.xz -o /tmp/zig.tar.xz && tar -xf /tmp/zig.tar.xz -C /tmp && sudo mv /tmp/zig-x86_64-linux-${ZIG_VERSION} /usr/local/zig && sudo ln -sf /usr/local/zig/zig /usr/local/bin/zig && rm /tmp/zig.tar.xz && zig version"
}

Commit .devcontainer/devcontainer.json to your GitHub branch, and if you already have an active codespace open,
trigger a Codespaces: Rebuild Container from the VS Code Command Palette (Ctrl+Shift+P / Cmd+Shift+P).
